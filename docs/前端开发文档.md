## 二、风格主题：拟态玻璃（Glassmorphism）

### 2.1 核心设计语言

设计目标：**通透、柔光、层次感** 的玻璃拟态风格，配合细腻的微动效，营造高级感。
支持 **亮色/暗色模式** 和 **多主题色切换**，所有颜色基于 CSS 自定义属性动态驱动。

### 2.2 主题系统（CSS 变量驱动）

> 根元素 `data-theme` 属性控制亮暗模式（`light` / `dark`），`data-primary-color` 控制主题色。
> 所有组件和 UnoCSS 快捷类都基于这些变量，切换主题无需重新编译。

#### 亮色模式变量

```css
:root,
[data-theme='light'] {
  /* ===== 玻璃背景 ===== */
  --glass-bg: rgba(255, 255, 255, 0.15);
  --glass-bg-strong: rgba(255, 255, 255, 0.25);
  --glass-bg-light: rgba(255, 255, 255, 0.08);

  /* ===== 玻璃导航 ===== */
  --nav-bg: rgba(255, 255, 255, 0.20);
  --nav-border: rgba(255, 255, 255, 0.30);
  --nav-shadow: 0 8px 32px rgba(31, 38, 135, 0.12);
  --nav-hover-bg: rgba(255, 255, 255, 0.30);

  /* ===== 边框 ===== */
  --glass-border: rgba(255, 255, 255, 0.25);

  /* ===== 阴影 ===== */
  --glass-shadow: 0 8px 32px 0 rgba(31, 38, 135, 0.15);
  --glass-shadow-lg: 0 12px 48px 0 rgba(31, 38, 135, 0.25);

  /* ===== 文字 ===== */
  --text-primary: #1a1a2e;
  --text-secondary: rgba(26, 26, 46, 0.65);
  --text-inverse: #f0f0f5;

  /* ===== 页面背景 ===== */
  --bg-page-start: #e0e5ec;
  --bg-page-end: #f5f7fa;
  --bg-page: linear-gradient(135deg, var(--bg-page-start), var(--bg-page-end));

  /* ===== 主题色（由 JS 动态覆写） ===== */
  --primary-color: #667eea;
  --primary-light: rgba(102, 126, 234, 0.15);
  --primary-hover: #5a6fd6;
  --primary-gradient: linear-gradient(135deg, var(--primary-color) 0%, #764ba2 100%);
  --primary-glow: 0 0 20px rgba(102, 126, 234, 0.3);
}
```

### 2.3 UnoCSS 预设配置

在 `uno.config.ts` 中配置自定义预设。所有颜色引用 CSS 变量以支持动态主题切换：

```typescript
// uno.config.ts
import { defineConfig, presetUno, presetAttributify } from 'unocss'

const themeColors = {
  primary: 'var(--primary-color)',
  primaryLight: 'var(--primary-light)',
  textPrimary: 'var(--text-primary)',
  textSecondary: 'var(--text-secondary)',
  textInverse: 'var(--text-inverse)',
  glass: 'var(--glass-bg)',
  glassStrong: 'var(--glass-bg-strong)',
  nav: 'var(--nav-bg)',
  pageStart: 'var(--bg-page-start)',
  pageEnd: 'var(--bg-page-end)',
}

export default defineConfig({
  presets: [presetUno(), presetAttributify()],
  shortcuts: {
    /* ===== 玻璃基础 ===== */
    'glass': 'bg-[var(--glass-bg)] backdrop-blur-4 border border-[var(--glass-border)] rounded-2xl shadow-lg',
    'glass-strong': 'bg-[var(--glass-bg-strong)] backdrop-blur-4 border border-[var(--glass-border)] rounded-2xl shadow-lg',
    'glass-hover': 'hover:-translate-y-0.5 hover:shadow-[var(--glass-shadow-lg)]',
    'glass-card': 'glass glass-hover transition-all duration-300 ease-out',

    /* ===== 悬浮气泡导航 ===== */
    'nav-bubble': 'bg-[var(--nav-bg)] backdrop-blur-5 border border-[var(--nav-border)] shadow-[var(--nav-shadow)] rounded-2xl',
    'nav-bubble-hover': 'hover:bg-[var(--nav-hover-bg)] transition-all duration-200',
    'nav-item': 'nav-bubble-hover cursor-pointer rounded-xl px-3 py-2 flex items-center gap-3',
    'nav-item-active': 'bg-[var(--primary-light)]! text-[var(--primary-color)]! rounded-xl',

    /* ===== 输入/按钮 ===== */
    'glass-input': 'bg-[var(--glass-bg)] backdrop-blur-2 border border-[var(--glass-border)] rounded-xl text-[var(--text-primary)]',
    'glass-btn': 'bg-[var(--glass-bg-strong)] backdrop-blur-2 border border-[var(--glass-border)] rounded-xl hover:bg-[var(--nav-hover-bg)] active:scale-[0.98] transition-all duration-200',

    /* ===== 渐变与背景 ===== */
    'primary-gradient': 'bg-gradient-to-br from-[var(--primary-color)] to-[#764ba2]',
    'page-bg': 'min-h-screen',
    'page-bg-light': 'bg-gradient-to-br from-[#e0e5ec] to-[#f5f7fa]',
    'page-bg-dark': 'bg-gradient-to-br from-[#0f0f1a] to-[#1a1a2e]',

    /* ===== 文字 ===== */
    'text-muted': 'text-[var(--text-secondary)]',
    'text-primary': 'text-[var(--text-primary)]',
    'text-accent': 'text-[var(--primary-color)]',
  },
  rules: [
    ['animate-float', { animation: 'float 6s ease-in-out infinite' }],
    ['animate-glow', { animation: 'glow 3s ease-in-out infinite alternate' }],
    ['animate-bubble-rise', { animation: 'bubble-rise 8s ease-in-out infinite' }],
    ['animate-bubble-float', { animation: 'bubble-float 4s ease-in-out infinite' }],
  ],
  preflights: [
    {
      getCSS: () => `
        @keyframes float {
          0%, 100% { transform: translateY(0px); }
          50% { transform: translateY(-10px); }
        }
        @keyframes glow {
          0% { box-shadow: 0 0 20px rgba(102, 126, 234, 0.3); }
          100% { box-shadow: 0 0 40px rgba(118, 75, 162, 0.5); }
        }
        @keyframes bubble-rise {
          0%, 100% { transform: translateY(0) scale(1); opacity: 0.6; }
          50% { transform: translateY(-20px) scale(1.05); opacity: 1; }
        }
        @keyframes bubble-float {
          0%, 100% { transform: translateY(0px) rotate(0deg); }
          33% { transform: translateY(-6px) rotate(1deg); }
          66% { transform: translateY(3px) rotate(-1deg); }
        }
        @keyframes fadeInUp {
          from { opacity: 0; transform: translateY(20px); }
          to { opacity: 1; transform: translateY(0); }
        }
        @keyframes slideInRight {
          from { opacity: 0; transform: translateX(30px); }
          to { opacity: 1; transform: translateX(0); }
        }
        @keyframes scaleIn {
          from { opacity: 0; transform: scale(0.95); }
          to { opacity: 1; transform: scale(1); }
        }
        @keyframes shimmer {
          0% { background-position: -200% 0; }
          100% { background-position: 200% 0; }
        }
        *, *::before, *::after {
          transition: background-color 0.3s ease, border-color 0.3s ease, box-shadow 0.3s ease, color 0.3s ease;
        }
        body {
          background: var(--bg-page);
          min-height: 100vh;
          color: var(--text-primary);
        }
      `
    }
```

## 三、项目初始化与命令

### 3.1 创建项目

```bash
cd code
pnpm create vite frontend --template vue-ts
cd frontend
```

### 3.2 安装依赖

```bash
# 核心依赖
pnpm add vue-router@4 pinia axios ant-design-vue@4 @ant-design/icons-vue
pnpm add @vueuse/core @vueuse/motion
pnpm add color  # 辅助色值计算（亮暗、透明度等）

# UnoCSS
pnpm add -D unocss @unocss/preset-uno @unocss/preset-attributify

# 开发依赖
pnpm add -D sass
```

### 3.3 开发命令

| 命令 | 说明 |
|------|------|
| `pnpm dev` | 启动开发服务器 (端口 18781) |
| `pnpm build` | 构建生产版本 |
| `pnpm preview` | 预览构建结果 |
| `pnpm lint` | ESLint 检查 |

### 3.4 Vite 配置

```typescript
// vite.config.ts
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import UnoCSS from 'unocss/vite'
import { resolve } from 'path'

export default defineConfig({
  plugins: [vue(), UnoCSS()],
  resolve: {
    alias: {
      '@': resolve(__dirname, 'src'),
    },
  },
  server: {
    port: 18781,
    proxy: {
      '/api': {
        target: 'http://localhost:18780',
        changeOrigin: true,
      },
    },
  },
})
```

### 5.2 路由守卫

```typescript
// src/router/guards.ts
import type { Router } from 'vue-router'
import { useUserStore } from '@/stores/user'

export function setupRouterGuard(router: Router) {
  router.beforeEach((to, _from, next) => {
    const userStore = useUserStore()
    const token = userStore.token

    // 已登录访问登录页 → 跳转到对应首页
    if (token && to.meta.guest) {
      return next(userStore.homePath)
    }

    // 未登录且不是公开页 → 跳登录
    if (!token && !to.meta.guest) {
      return next('/login')
    }

    // 已登录但角色不匹配 → 跳对应首页
    if (token && to.meta.role && to.meta.role !== userStore.role) {
      return next(userStore.homePath)
    }

    // 已登录但状态异常（待审核/冻结）→ 登出并提示
    if (token && userStore.status !== 2 && !to.meta.guest) {
      if (userStore.status === 1) {
        ElMessage.warning('账号待审核，请联系管理员')
      } else if (userStore.status === 3) {
        ElMessage.error('账号已被冻结')
      }
      userStore.logout()
      return next('/login')
    }

    next()
  })
}
```

角色首页映射：

| 角色 | `homePath` |
|------|-----------|
| 管理员 (1) | `/admin/shops` |
| 店铺用户 (2) | `/operator/shop` |
| 普通用户 (3) | `/user/products` |

## 九、API 请求规范

### 9.1 Axios 实例

```typescript
// src/api/request.ts
import axios from 'axios'
import { message } from 'ant-design-vue'
import { useUserStore } from '@/stores/user'
import router from '@/router'

const request = axios.create({
  baseURL: '/api',
  timeout: 15000,
})

// 请求拦截器：自动附带 Token
request.interceptors.request.use((config) => {
  const userStore = useUserStore()
  if (userStore.token) {
    config.headers.Authorization = `Bearer ${userStore.token}`
  }
  return config
})

// 响应拦截器：统一错误处理
request.interceptors.response.use(
  (response) => {
    const res = response.data
    if (res.code !== 200) {
      message.error(res.message || '请求失败')
      return Promise.reject(new Error(res.message))
    }
    return res
  },
  (error) => {
    if (error.response?.status === 401) {
      const userStore = useUserStore()
      userStore.logout()
      router.push('/login')
      message.error('登录已过期，请重新登录')
    } else {
      message.error(error.response?.data?.message || error.message || '网络异常')
    }
    return Promise.reject(error)
  },
)

export default request
```

### 9.4 错误处理守则

> **核心原则：禁止吞没错误。** 任何 `try-catch` 或 `.catch()` 块中，捕获到错误后必须 `throw` 或 `return Promise.reject()` 将错误继续向上传递，让调用方能够感知并排查失败原因。

#### ❌ 禁止写法

```typescript
// 错误：捕获错误后什么都不做，代码会"静默"继续执行
try {
  const res = await api.getUserInfo()
  userInfo.value = res.data
} catch (e) {
  // 什么都没做 —— 请求失败了用户却不知情，后续代码可能在残缺数据下运行
}

// 错误：.catch 后无任何处理
api.getUserInfo()
  .then(res => { userInfo.value = res.data })
  .catch(() => {})  // 错误被吞掉

// 错误：catch 后返回一个"假"值，导致调用方误以为成功
try {
  const res = await api.getUserInfo()
  return res.data
} catch {
  return null  // 调用方收到 null 无法区分是正常空值还是请求失败
}
```

#### ✅ 正确写法

```typescript
// 正确：捕获后重新抛出，错误继续向上传递
try {
  const res = await api.getUserInfo()
  userInfo.value = res.data
} catch (e) {
  // 可以根据需要先做日志或局部处理，但最终必须 throw
  console.error('[getUserInfo] 请求失败:', e)
  throw e  // 必须抛出，让调用方感知错误
}

// 正确：.catch 中 return Promise.reject
api.getUserInfo()
  .then(res => { userInfo.value = res.data })
  .catch(e => {
    console.error('[getUserInfo] 请求失败:', e)
    return Promise.reject(e)
  })

// 正确：仅在明确不需要向上传递的场景（如"非关键接口"），也必须加日志
// 但这类场景极少，默认都应该 throw
```

#### 特别提醒

- **请求层（API 调用）**：响应拦截器已统一 `return Promise.reject(error)`，业务代码无需再重复处理通用错误（如 401）。但业务层 `try-catch` 依然**必须**将意外错误重新抛出。
- **UI 事件处理**：按钮点击、表单提交等用户交互中的请求失败，同样禁止吞没错误，至少应在控制台输出错误信息并给用户 Toast 提示。
- **错误边界组件**：如需捕获渲染层错误，使用 Vue 的 `onErrorCaptured` 或全局错误处理，而非在业务代码中 `try-catch` 后不抛出。

### 10.3 主题状态（亮暗 + 主题色）

```typescript
// src/stores/theme.ts
import { defineStore } from 'pinia'
import { ref, watch } from 'vue'
import Color from 'color'

type ThemeMode = 'light' | 'dark' | 'auto'

interface ThemeState {
  mode: ThemeMode
  primaryColor: string
}

const DEFAULT_COLOR = '#667eea'
const STORAGE_KEY = 'theme-config'

// 从 localStorage 恢复
function loadTheme(): ThemeState {
  try {
    const saved = localStorage.getItem(STORAGE_KEY)
    if (saved) return JSON.parse(saved)
  } catch { /* ignore */ }
  return { mode: 'light', primaryColor: DEFAULT_COLOR }
}

// 将颜色转为 rgba 变体
function expandColor(hex: string) {
  const c = Color(hex)
  return {
    base: hex,
    light: c.alpha(0.15).toString(),
    hover: c.darken(0.08).hex(),
    gradient: `linear-gradient(135deg, ${hex} 0%, ${c.darken(0.2).hex()} 100%)`,
    glow: `0 0 20px ${c.alpha(0.3).toString()}`,
  }
}

export const useThemeStore = defineStore('theme', () => {
  const saved = loadTheme()
  const mode = ref<ThemeMode>(saved.mode)
  const primaryColor = ref(saved.primaryColor)

  // 实际生效的模式（auto 时跟随系统）
  const effectiveMode = ref<'light' | 'dark'>(mode.value === 'auto'
    ? getSystemMode()
    : mode.value
  )

  // 应用主题到 document
  function apply() {
    const actualMode = mode.value === 'auto' ? getSystemMode() : mode.value
    effectiveMode.value = actualMode

    document.documentElement.setAttribute('data-theme', actualMode)
    document.documentElement.style.setProperty('--primary-color', primaryColor.value)

    const expanded = expandColor(primaryColor.value)
    document.documentElement.style.setProperty('--primary-light', expanded.light)
    document.documentElement.style.setProperty('--primary-hover', expanded.hover)
    document.documentElement.style.setProperty('--primary-gradient', expanded.gradient)
    document.documentElement.style.setProperty('--primary-glow', expanded.glow)

    // 保存到 localStorage
    localStorage.setItem(STORAGE_KEY, JSON.stringify({
      mode: mode.value,
      primaryColor: primaryColor.value,
    }))
  }

  function setMode(m: ThemeMode) {
    mode.value = m
    apply()
  }

  function setPrimaryColor(color: string) {
    primaryColor.value = color
    apply()
  }

  function reset() {
    mode.value = 'light'
    primaryColor.value = DEFAULT_COLOR
    apply()
  }

  // 监听系统颜色方案变化
  let mediaQuery: MediaQueryList | null = null
  function init() {
    apply()
    mediaQuery = window.matchMedia('(prefers-color-scheme: dark)')
    mediaQuery.addEventListener('change', () => {
      if (mode.value === 'auto') apply()
    })
  }
  function destroy() {
    mediaQuery?.removeEventListener('change', () => {})
  }

  return { mode, primaryColor, effectiveMode, setMode, setPrimaryColor, reset, init, destroy }
})

function getSystemMode(): 'light' | 'dark' {
  return window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light'
}
```
